# Passwortfreies Herunterfahren fuer IntruderWatch einrichten.
# EINMALIG im Terminal ausfuehren (fragt dein Admin-Passwort):

echo "$(whoami) ALL=(root) NOPASSWD: /sbin/shutdown" | sudo tee /etc/sudoers.d/intruderwatch-shutdown >/dev/null && sudo chmod 440 /etc/sudoers.d/intruderwatch-shutdown && echo OK

# Pruefen danach:
sudo -n /sbin/shutdown -h +60   # plant Shutdown in 60 min OHNE Passwort -> dann abbrechen:
sudo killall shutdown   # (macOS kennt kein 'shutdown -c')
