# Set up password-less shutdown for IntruderWatch.
# Run ONCE in Terminal (asks for your admin password):

echo "$(whoami) ALL=(root) NOPASSWD: /sbin/shutdown" | sudo tee /etc/sudoers.d/intruderwatch-shutdown >/dev/null && sudo chmod 440 /etc/sudoers.d/intruderwatch-shutdown && echo OK

# Check afterwards:
sudo -n /sbin/shutdown -h +60   # schedules a shutdown in 60 min WITHOUT a password -> then cancel:
sudo killall shutdown   # (macOS has no 'shutdown -c')
